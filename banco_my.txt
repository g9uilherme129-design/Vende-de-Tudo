CREATE DATABASE sistema_vendas;
USE sistema_vendas;

-- tabela usuário
CREATE TABLE IF NOT EXISTS usuario(
	id_user VARCHAR(7) PRIMARY KEY ,
	nome_user VARCHAR(100) NOT NULL,
    cpf VARCHAR(11) NOT NULL UNIQUE,
	email_user VARCHAR(100) NOT NULL UNIQUE,
	senha_user VARCHAR(66) NOT NULL,
	status_user BOOLEAN DEFAULT TRUE,
	perfil VARCHAR(9) NOT NULL,
    salario DECIMAL(10,2) NOT NULL,
    motivo_desat VARCHAR(30) NULL,
    data_desat DATETIME NULL,
    admin_desat VARCHAR(7) NULL,
    tema_dark TINYINT(1) DEFAULT 1,
    pontuacao_maxima INT DEFAULT 0
);

 -- tabela categoria
 CREATE TABLE IF NOT EXISTS categoria(
	id_categoria VARCHAR(7) PRIMARY KEY ,
    nome_categoria VARCHAR(150) NOT NULL UNIQUE,
    tipo_categoria VARCHAR(20) NOT NULL,
    marca VARCHAR(30) NOT NULL,
    status_categoria BOOLEAN DEFAULT TRUE
 );

-- tabela fornecedor
CREATE TABLE IF NOT EXISTS fornecedor(
	id_fornecedor VARCHAR(7) PRIMARY KEY,
    nome_fornecedor VARCHAR(150) NOT NULL UNIQUE,
    CNPJ VARCHAR(14) NOT NULL UNIQUE,
    telefone VARCHAR(12) NOT NULL,
    email_forn varchar(100) NOT NULL UNIQUE,
    endereco_logradouro VARCHAR(150) NOT NULL,
    endereco_numero VARCHAR(20) NOT NULL,
    bairro VARCHAR(100) NOT NULL,
    cidade VARCHAR(100) NOT NULL,
    estado VARCHAR(100) NOT NULL,
    cep VARCHAR(10) NOT NULL,
    status_fornecedor VARCHAR(10) DEFAULT 'Ativo'
);

ALTER TABLE fornecedor MODIFY COLUMN status_fornecedor VARCHAR(10) DEFAULT 'Ativo';

-- tabela estoque
CREATE TABLE IF NOT EXISTS estoque(
	id_estoque VARCHAR(7) PRIMARY KEY,
    id_fornecedor VARCHAR(7) NOT NULL,
    id_categoria VARCHAR(7) NOT NULL,
    nome_estoque VARCHAR(100) NOT NULL,
    codigo_barras varchar(10) NOT NULL UNIQUE,
    data_validade DATE NOT NULL,
    data_entrada DATE NOT NULL,
    preco_unitario DECIMAL(10,2) NOT NULL,
    preco_venda DECIMAL(10,2) NOT NULL,
    embalagem VARCHAR(30) NOT NULL,
    quantidade int UNSIGNED DEFAULT 0,
    lote VARCHAR(9) NOT NULL,
    
	FOREIGN KEY (id_fornecedor) REFERENCES fornecedor(id_fornecedor),
	FOREIGN KEY (id_categoria) REFERENCES categoria(id_categoria)
);

-- tabela vendas
CREATE TABLE IF NOT EXISTS venda(
	id_venda VARCHAR(7) PRIMARY KEY,
    id_user VARCHAR(7) NOT NULL,
    id_estoque VARCHAR(7) NOT NULL,
    metodo_pagamento VARCHAR(20) NOT NULL,
    data_venda DATETIME NOT NULL,
    quantidade INT NOT NULL,
    qtd_venda INT NOT NULL,
    
    FOREIGN KEY (id_user) REFERENCES usuario(id_user),
    FOREIGN KEY (id_estoque) REFERENCES estoque(id_estoque)
);
